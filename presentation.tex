\section{Introduction}

\begin{frame}{Problem Statement}
  \textbf{Current practice:} Property valuation in India relies on local
  brokers, recent comparable sales and subjective judgment.

  \medskip
  \textbf{Limitations:}
  \begin{itemize}
    \item Inconsistent quality across valuers
    \item Slow turnaround
    \item Susceptible to bias
    \item Difficult to scale across 20+ states
  \end{itemize}

  \medskip
  \textbf{Goal:} Build an automated, data-driven valuation tool that delivers
  fast, consistent and accurate price estimates for the Indian market.
\end{frame}

\begin{frame}{Objectives}
  \begin{enumerate}
    \item \textbf{Data pipeline} --- preprocessing and feature engineering for
          Indian housing records.
    \item \textbf{Multi-model training} --- XGBoost, LightGBM (GPU) and Random
          Forest on a common train/val/test split.
    \item \textbf{Weighted ensemble} --- composite performance scores to combine
          models robustly.
    \item \textbf{Confidence output} --- price range with High / Medium / Low
          indicator.
    \item \textbf{Web deployment} --- Flask REST API with a browser UI.
    \item \textbf{Accuracy target} --- R\textsuperscript{2} $\ge$ 0.85,
          MAPE $<$ 10\%.
  \end{enumerate}
\end{frame}

\begin{frame}{Scope and Limitations}
  \textbf{In scope:}
  \begin{itemize}
    \item Residential properties
    \item 20 Indian states, 42 cities
  \end{itemize}

  \textbf{Out of scope:}
  \begin{itemize}
    \item Commercial / agricultural properties
    \item Macroeconomic factors (interest rates, inflation)
    \item Temporal trends and market cycles
  \end{itemize}

  \medskip
  \emph{Predictions are estimates for decision support, not formal appraisals.}
\end{frame}

\section{Methodology}

\begin{frame}{System Architecture}
  \begin{enumerate}
    \item \textbf{Data layer} --- CSV ingestion and schema validation
    \item \textbf{Preprocessing} --- cleaning, one-hot encoding, feature
          engineering
    \item \textbf{Scaling} --- MinMaxScaler $\to [0,1]$
    \item \textbf{Training} --- XGBoost, LightGBM (GPU), Random Forest
    \item \textbf{Ensemble} --- composite scoring $\to$ weighted average
          $\to$ price range
    \item \textbf{Deployment} --- Flask REST API + web interface
  \end{enumerate}
\end{frame}

\begin{frame}{Preprocessing Pipeline}
  \begin{enumerate}
    \item \textbf{Deduplication} --- remove duplicate rows
    \item \textbf{Outlier censoring} --- clip to [0.2, 99.8] percentiles
    \item \textbf{Null handling} --- drop rows with missing values
    \item \textbf{Amenity parsing} --- \texttt{Amenities} $\to$
          \texttt{amenity\_count}
    \item \textbf{Encoding} --- binary $\to$ \{0,1\}, categorical $\to$ one-hot
    \item \textbf{Split} --- 70/10/20 with \texttt{random\_state=42}
  \end{enumerate}
\end{frame}

\begin{frame}{Models --- XGBoost}
  \textbf{Approach:} Sequential gradient boosting with histogram-based trees.

  \medskip
  \textbf{Configuration:}
  \begin{itemize}
    \item \texttt{max\_depth=6},\; \texttt{learning\_rate=0.03}
    \item \texttt{subsample=0.8},\; \texttt{colsample\_bytree=0.8}
    \item \texttt{reg\_alpha=0.1},\; \texttt{reg\_lambda=1.0}
    \item \texttt{tree\_method='hist'}
  \end{itemize}

  \medskip
  \textbf{Result:} Early stopping at 268 trees.
\end{frame}

\begin{frame}{Models --- LightGBM (GPU)}
  \textbf{Approach:} Leaf-wise gradient boosting with GPU acceleration.

  \medskip
  \textbf{Configuration:}
  \begin{itemize}
    \item \texttt{max\_depth=6},\; \texttt{learning\_rate=0.03}
    \item \texttt{num\_leaves=63},\; \texttt{subsample=0.8}
    \item \texttt{colsample\_bytree=0.8}
    \item \texttt{device='gpu'} (NVIDIA RTX 3060, 6 GB VRAM)
  \end{itemize}

  \medskip
  \textbf{Result:} Early stopping at 365 trees, training $\approx$ 16 seconds.
\end{frame}

\begin{frame}{Models --- Random Forest}
  \textbf{Approach:} Bootstrap aggregation with random feature selection.

  \medskip
  \textbf{Configuration:}
  \begin{itemize}
    \item \texttt{n\_estimators=300},\; \texttt{max\_depth=25}
    \item \texttt{min\_samples\_split=5},\; \texttt{min\_samples\_leaf=2}
    \item \texttt{max\_features=1.0},\; \texttt{n\_jobs=-1}
  \end{itemize}

  \medskip
  \textbf{Result:} Training on CPU $\approx$ 6 minutes.
\end{frame}

\begin{frame}{Weighted Ensemble --- Scoring}
  Each model $i$ receives a composite score $S_i$ based on three test-set metrics:

  \medskip
  \[
  S_i = \frac{ R^2_i \;+\; \dfrac{1}{1+\mathrm{MAE}_i/100} \;+\;
         \dfrac{1}{1+\mathrm{MAPE}_i/100} }{3}
  \]

  \medskip
  Scores are normalised to sum to one, producing weights $w_i$.

  \[
  \hat{y} = \frac{\sum w_i \cdot p_i}{\sum w_i}
  \]

  \medskip
  \textbf{Price range:} $\hat{y} \pm 1.5\sigma$, where $\sigma$ is the
  standard deviation of the three individual predictions.

  \medskip
  \textbf{Confidence:} High if $\sigma < 20$ Lakhs, Medium if
  $20\le\sigma<50$, Low otherwise.
\end{frame}

\section{Conclusion}

\begin{frame}{Summary of Achievements}
  \begin{itemize}
    \item \textbf{Performance:} Ensemble $R^2 = 0.9962$, MAPE $\approx 5.0\%$
          --- far beyond the $R^2 \ge 0.85$ target.
    \item \textbf{GPU training:} LightGBM trains in $\approx 16$ seconds on an
          RTX 3060.
    \item \textbf{Robust ensemble:} Weighted composite scoring produces
          predictions that are more reliable than any single model.
    \item \textbf{Confidence output:} Price range + confidence indicator
          gives users actionable information.
    \item \textbf{Pan-India:} Trained on data from 20 states and 42 cities.
    \item \textbf{Production-ready:} Flask web service with auto-filled city
          rates minimises user input.
  \end{itemize}
\end{frame}
